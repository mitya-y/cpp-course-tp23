\begin{frame}[fragile]{Передача как callback аргумент}
\begin{minted}[fontsize=\small]{go}
package main

import "fmt"
import "slices"

func main() {
  words := []string{"aa", "aaa", "a"}
  slices.SortFunc(words, func(a, b string) int {
      return len(a) - len(b)
  })
  fmt.Println(words) // [a aa aaa]
}

\begin{frame}[fragile]{Замыкания}
\begin{minted}[fontsize=\small]{go}
package main
import "fmt"
func get_generator() func() int {
  i := 0 // 'i' will copy by reference
  return func() int {
    i++
    return i
  }
}
func main() {
  a := get_generator()
  fmt.Println(a()) // 1
  fmt.Println(a()) // 2

  b := get_generator()
  fmt.Println(b()) // 1
}
\end{minted}
\end{frame}

\begin{frame}[fragile]{Замыкания - передача по копии или ссылке?}
  \begin{columns}[T, onlytextwidth]
    \setlength{\columnsep}{0pt}
    \begin{column}{0.47\textwidth}
      \begin{minted}[fontsize=\footnotesize]{go}
package main
import "fmt"
func call(f func(bool) int) { f(true) }
func main() {
  i := 0
  // 'i' will copy by reference
  f := func(inc bool) int {
    if inc { i++ }
    return i
  }
  g := func(val int) func(bool) int {
    return func(inc bool) int {
      if inc { val++ }
      return val
    }
  }(i)
      \end{minted}
    \end{column}
    \begin{column}{0.53\textwidth}
      \begin{minted}[fontsize=\footnotesize]{go}
  call(f)
  fmt.Println(i, f(false), g(false)) // 1, 1, 0
  call(g)
  fmt.Println(i, f(false), g(false)) // 1, 1, 1
}
      \end{minted}
    \end{column}
  \end{columns}
\end{frame}
